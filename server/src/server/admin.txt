// Panel auth. The key is exchanged ONCE for an HttpOnly session cookie, so it
// never lives in JS-reachable storage afterwards (an XSS can act as the user
// but cannot read the key out). State-changing calls must echo the CSRF token
// the login returned; the server rejects cookie-authenticated writes without it.
let CSRF = sessionStorage.getItem("zk-csrf") || "";

/**
 * Whether the OPERATOR has done something recently.
 *
 * The panel polls on timers (/overview every 5s, logs every 3s, remote every
 * 4s), so "a request happened" says nothing about whether anyone is there. If
 * those polls extended the session the idle timeout could never fire — the
 * panel would stay signed in indefinitely with the tab merely open.
 *
 * So activity is tracked from real input events and reported as a header only
 * while it is fresh. Once the operator stops touching the page the flag lapses
 * and the polls go back to being non-extending.
 */
let lastInputAt = Date.now();
for (const ev of ["pointerdown", "keydown", "wheel", "touchstart"]) {
  addEventListener(ev, () => { lastInputAt = Date.now(); }, { passive: true, capture: true });
}
/** True when the operator interacted within the last 30 seconds. */
function userActive() {
  return Date.now() - lastInputAt < 30000;
}

/**
 * True only when a 401/403 means the PANEL session lapsed.
 *
 * The panel used to treat every 401/403 as "your session expired", which is
 * wrong once the admin API also fronts the proxy: a request the UPSTREAM refused
 * comes back through the same channel. Concretely — a risk-control block (3012)
 * is answered with 403 `model_blocked_upstream`, so sending a message while
 * upstream was blocking popped the login dialog every single time. It reads as
 * "sending a chat logs me out", and it sent people looking for a session bug
 * that was never there.
 *
 * Our own session failures have exactly two shapes, and both are distinguishable
 * from a proxy error by the `error.type` we already send:
 *   - 401 `unauthorized`  — no/again-invalid admin key
 *   - 403 `csrf_failed`   — the CSRF token was cleared by a failed attempt
 * Anything else is passed through to the caller untouched.
 */
async function isSessionLapse(resp) {
  if (resp.status !== 401 && resp.status !== 403) return false;
  let type = "";
  try {
    const body = await resp.clone().json();
    type = (body && body.error && body.error.type) || "";
  } catch (e) {
    // Not a JSON error body. Fall through to the conservative answer below.
  }
  // An unrecognised shape re-gates, which is the pre-existing behaviour: a real
  // lapse must never dead-end the UI. Only a KNOWN non-session type is passed
  // through, so this stays correct against an older engine too.
  if (!type) return true;
  return type === "unauthorized" || type === "csrf_failed";
}

async function api(path, opts) {
  const method = (opts && opts.method) || "GET";
  const headers = { "content-type": "application/json" };
  if (CSRF && method !== "GET" && method !== "HEAD") headers["x-csrf-token"] = CSRF;
  // Tells the server to slide the idle window. A write is inherently an action,
  // so it always counts; a read only counts when the operator has been active.
  if (userActive() || (method !== "GET" && method !== "HEAD")) headers["x-zk-active"] = "1";
  const resp = await fetch("/admin/api" + path, { ...opts, headers, credentials: "same-origin" });
  if (await isSessionLapse(resp)) {
    // The session lapsed (cookie expired, engine restarted, or the CSRF token
    // was cleared). Re-gate the UI and ask again — the same path the initial
    // load uses, so there is one login flow rather than two.
    return regateAndRetry(path, opts);
  }
  return resp;
}

/**
 * Re-authenticate after a lapsed session, then replay the request once.
 *
 * Guarded by a single in-flight promise: several panel calls fire on a timer,
 * and without this each would raise its own dialog.
 */
let regatePromise = null;
async function regateAndRetry(path, opts) {
  if (!regatePromise) {
    document.body.classList.add("gated");
    regatePromise = (async () => {
      const key = await askAdminKey();
      return key ? await doLogin(key) : false;
    })().finally(() => { regatePromise = null; });
  }
  const ok = await regatePromise;
  if (!ok) return new Response(null, { status: 401 });
  document.body.classList.remove("gated");
  // CSRF was cleared by the failed attempt; drop it so doLogin's fresh token is used.
  const method = (opts && opts.method) || "GET";
  const headers = { "content-type": "application/json" };
  if (CSRF && method !== "GET" && method !== "HEAD") headers["x-csrf-token"] = CSRF;
  return fetch("/admin/api" + path, { ...opts, headers, credentials: "same-origin" });
}

/** Exchange a key for a session cookie. Returns true on success. */
async function doLogin(key) {
  try {
    const r = await fetch("/admin/api/login", {
      method: "POST",
      headers: { "content-type": "application/json" },
      credentials: "same-origin",
      body: JSON.stringify({ key }),
    });
    const d = await r.json().catch(() => ({}));
    if (!r.ok) {
      // Surface the server's own reason: a lockout and a wrong key need
      // different reactions from the user.
      const msg = (d.error && d.error.message) || ("HTTP " + r.status);
      toast(msg);
      return false;
    }
    CSRF = d.csrf || "";
    sessionStorage.setItem("zk-csrf", CSRF);
    return true;
  } catch (e) { toast(String(e)); return false; }
}

/**
 * Show the built-in key dialog and resolve with the entered value (or null on
 * cancel). One dialog at a time: concurrent 401s share the same promise so the
 * user is not asked twice.
 */
let keyDialogPromise = null;
function askAdminKey() {
  if (keyDialogPromise) return keyDialogPromise;
  keyDialogPromise = new Promise((resolve) => {
    const mask = document.getElementById("key-mask");
    const input = document.getElementById("key-input");
    const err = document.getElementById("key-err");
    input.value = "";
    err.style.display = "none";
    mask.classList.add("on");
    setTimeout(() => input.focus(), 50);
    const done = (val) => {
      mask.classList.remove("on");
      document.getElementById("key-ok").onclick = null;
      document.getElementById("key-cancel").onclick = null;
      input.onkeydown = null;
      keyDialogPromise = null;
      resolve(val);
    };
    document.getElementById("key-ok").onclick = () => {
      const v = input.value.trim();
      if (!v) { err.style.display = "block"; return; }
      done(v);
    };
    document.getElementById("key-cancel").onclick = () => done(null);
    input.onkeydown = (e) => {
      if (e.key === "Enter") { e.preventDefault(); document.getElementById("key-ok").click(); }
      if (e.key === "Escape") { e.preventDefault(); done(null); }
    };
  });
  return keyDialogPromise;
}
function toast(msg) {
  const t = document.getElementById("toast");
  t.textContent = msg; t.classList.add("on");
  clearTimeout(t._tm); t._tm = setTimeout(() => t.classList.remove("on"), 2600);
}

/**
 * In-page confirmation, replacing the browser's native confirm().
 *
 * Returns a Promise<boolean>, so a caller reads:
 *     if (!(await askConfirm(msg))) return;
 * — the same shape as the native call, which keeps every conversion a
 * one-line change and makes the async-ness obvious at the call site.
 *
 * Why not the native dialog: it is browser chrome rather than part of the
 * panel, it blocks the entire tab, and a browser setting can suppress it — in
 * which case `confirm()` returns false and a destructive action becomes
 * silently unreachable, or (worse, in other implementations) true and
 * unconfirmed. Neither failure mode is acceptable for "delete this account".
 *
 * Only one dialog exists, so a second call while one is open would strand the
 * first promise. That cannot happen in this panel — the dialog is modal and
 * blocks the UI — but the guard makes it impossible rather than unlikely.
 */
let askResolve = null;
function askConfirm(body, opts) {
  const o = opts || {};
  const mask = document.getElementById("ask-mask");
  document.getElementById("ask-title").textContent = o.title || T("confirm_title");
  document.getElementById("ask-body").textContent = body || "";
  const okBtn = document.getElementById("ask-ok");
  // A destructive action gets the danger styling and its own verb, so the
  // button says what it does rather than a generic "OK".
  okBtn.className = "btn" + (o.danger ? " danger" : "");
  okBtn.textContent = o.okText || T("ok");
  mask.classList.add("on");
  okBtn.focus();
  return new Promise((resolve) => {
    askResolve = resolve;
  });
}
function askClose(result) {
  document.getElementById("ask-mask").classList.remove("on");
  const r = askResolve;
  askResolve = null;
  if (r) r(result);
}
document.getElementById("ask-ok").onclick = () => askClose(true);
document.getElementById("ask-cancel").onclick = () => askClose(false);
// Clicking the backdrop cancels, matching every other dialog in the panel.
document.getElementById("ask-mask").addEventListener("click", (e) => {
  if (e.target.id === "ask-mask") askClose(false);
});
// Esc cancels and Enter confirms, so the dialog is usable without a mouse.
addEventListener("keydown", (e) => {
  if (!document.getElementById("ask-mask").classList.contains("on")) return;
  if (e.key === "Escape") { e.preventDefault(); askClose(false); }
  if (e.key === "Enter") { e.preventDefault(); askClose(true); }
});
const fmtNum = n => n >= 1e8 ? (n / 1e8).toFixed(1) + "亿" : n >= 1e4 ? Math.round(n / 1e4) + "万" : String(n);

/**
 * Switch to a page, wiring up whatever that page needs to load.
 *
 * Split out of the click handler so the initial page can be restored from the
 * last visit without a synthetic click — a click would run the nav button's own
 * bookkeeping twice and briefly flash the wrong page.
 */
function showPage(name) {
  const btn = document.querySelector('#nav button[data-page="' + name + '"]');
  const page = document.getElementById("page-" + name);
  if (!btn || !page) return false;             // unknown page: keep the current one
  document.querySelectorAll("#nav button").forEach(b => b.classList.remove("on"));
  document.querySelectorAll(".page").forEach(p => p.classList.remove("on"));
  btn.classList.add("on");
  page.classList.add("on");
  if (name === "playground") { pgRestore(); loadPlaygroundAccounts(); }
  if (name === "accounts") refreshAccounts();
  if (name === "models") loadProbeAccounts();
  if (name === "logs") pollLogs();
  if (name === "settings") loadExportSummary();
  return true;
}

/** Remember the current page so a refresh comes back to it, not to Overview. */

}

async function refreshOverview() {
  try {
    const r = await api("/overview");
    const ov = await r.json();
    document.getElementById("ov-total").textContent = ov.accounts.total;
    document.getElementById("ov-eligible").textContent = ov.accounts.eligible;
    document.getElementById("ov-inflight").textContent = ov.accounts.inFlight;

    // still eligible (it serves its other models), so without this the header

    const thr = document.getElementById("ov-throttled");

    document.getElementById("ov-reqs").textContent = (ov.requests && ov.requests.total) || 0;

      document.getElementById("ov-ttfb").textContent = fmtMs(ov.requests.avgTtfbMs);
      document.getElementById("ov-okfail").textContent = ov.requests.succeeded + " / " + ov.requests.failed;
      document.getElementById("ov-tokens").textContent = fmtNum(ov.requests.tokensTotal);
    }
    renderRecent(ov.recent || []);
  } catch {}
}

/** Format a millisecond duration compactly (0 → "0ms"). */
function fmtMs(ms) {
  if (!ms) return "0ms";
  if (ms < 1000) return ms + "ms";
  return (ms / 1000).toFixed(1) + "s";
}

/**
 * Compress a User-Agent to the part that identifies the client.
 *
 * Lives here rather than beside the request table because the merged log
 * renders request rows too. Deleting it with the old table code left
 * `renderMergedLog` throwing a ReferenceError that the surrounding try/catch
 * swallowed, so the page showed an empty log with no visible error at all.
 */
function shortUa(ua) {
  if (!ua) return "—";
  const m = /^(claude-cli|claude-code|codex|openai|anthropic|curl|python|node|axios|okhttp|Go-http)[^\s]*/i.exec(ua);
  if (m) return m[1];
  // Fall back to the first product token, which is the client's own name.
  return ua.split(/[\s/]/)[0].slice(0, 24);
}

/**
 * One-line request summary for the overview.
 *
 * Deliberately NOT a table: the Logs page shows the same requests with more
 * columns (caller IP, path, client) and can filter and export them. Repeating
 * the table here duplicated the data and made the overview long enough to push
 * the balance cards off screen.
 */
function renderRecent(rows) {
  const el = document.getElementById("ov-recent");
  if (!el) return;
  if (!rows.length) {
    el.innerHTML = '<span class="bal-empty">' + T("ov_recent_none") + "</span>";
    return;
  }
  const last = rows[0];
  const okCount = rows.filter((r) => r.status >= 200 && r.status < 400).length;
  const rate = Math.round((okCount / rows.length) * 100);
  const parts = [
    T("ov_recent_last").replace("{n}", String(rows.length)),
    T("ov_recent_rate").replace("{p}", String(rate)),
  ];
  if (last) {
    parts.push(T("ov_recent_latest")
      .replace("{t}", new Date(last.started).toLocaleTimeString())
      .replace("{m}", last.model || "—"));
  }
  el.innerHTML = esc(parts.join(" · "));
}
/** Cached quota payload so the search box can re-render without re-fetching. */
let lastQuota = [];
let lastAccountMeta = new Map();

async function loadBalances(knownAccounts, knownQuota) {
  const el = document.getElementById("ov-balances");
  // Show the placeholder only when nothing has been rendered yet.
  //
  // This used to blank the list on EVERY call, and refreshAccounts() calls it on
  // every pool change — so refreshing on the ACCOUNTS page wiped the OVERVIEW's
  // finished list back to "Loading…" and repainted it from scratch. That is the
  // hitch seen on the overview after touching the accounts page, and it bought
  // nothing: the data was about to arrive either way.
  if (el.dataset.loaded !== "1") {
    el.innerHTML = '<span class="sub">' + T("loading") + "</span>";
  }
  try {
    // `knownAccounts` / `knownQuota`: the caller already holds them (see
    // refreshAccounts), so re-reading either would be a second round trip for
    // data already in hand.
    const [data, accs] = await Promise.all([
      knownQuota ? Promise.resolve(knownQuota) : api("/accounts/quota").then((r) => r.json()),
      knownAccounts
        ? Promise.resolve(knownAccounts)
        : api("/accounts").then((r) => r.json()).then((d) => d.accounts || []),
    ]);
    lastQuota = data.accounts || [];
    lastAccountMeta = new Map(accs.map(a => [a.id, a]));
    // Marks the overview as rendered, so refreshAccounts knows a later pool
    // change has something to update (see refreshOverviewBalances).
    el.dataset.loaded = "1";
    renderTotals(lastQuota);
    renderBalanceList();
  } catch {
    el.innerHTML = '<span class="sub">—</span>';
    document.getElementById("ov-total-quota").textContent = "—";
  }
}

/** Pool-wide totals, grouped by model so "how much do I have left" is one number. */
function renderTotals(list) {
  const byModel = new Map();
  let accountCount = 0;
  for (const a of list) {
    if ((a.balances || []).length) accountCount++;
    for (const b of a.balances || []) {
      const cur = byModel.get(b.showName) || { remaining: 0, total: 0, accounts: new Set() };
      cur.remaining += b.remainingUnits;
      cur.total += b.totalUnits;
      cur.accounts.add(a.id);
      byModel.set(b.showName, cur);
    }
  }
  const chips = document.getElementById("ov-total-chips");
  chips.innerHTML = "";
  const entries = [...byModel.entries()].sort((x, y) => y[1].remaining - x[1].remaining);
  let grand = 0;
  for (const [model, v] of entries) {
    grand += v.remaining;
    const c = document.createElement("div");
    c.className = "tot-chip";
    c.innerHTML = '<div class="c1">' + esc(model) + " · " + v.accounts.size + T("acc_unit") + "</div>" +
      '<div class="c2">' + fmtNum(v.remaining) + "</div>";
    chips.appendChild(c);
  }
  document.getElementById("ov-total-quota").textContent = entries.length ? fmtNum(grand) : "—";
  document.getElementById("ov-total-sub").textContent = entries.length
    ? T("ov_total_sub").replace("{n}", String(accountCount)).replace("{m}", String(entries.length))
    : T("ov_total_none");
}

/**
 * Explain an empty balance row by its actual cause.
 *
 * Four different situations produce zero balance rows, and they call for four
 * different operator actions — re-login, wait, subscribe, or nothing at all.
 * The engine reports which one via `balanceState`; this picks the matching line.
 * An upstream error string wins when there is one, because it carries detail the
 * fixed label cannot ("connect timeout" beats "read failed").
 */
function balanceEmptyText(rec) {
  if (rec && rec.errors && rec.errors.length) return String(rec.errors[0]).slice(0, 90);
  const state = rec && rec.balanceState;
  if (state === "no_plan") return T("bal_no_plan");
  if (state === "unavailable") return T("bal_unavailable");
  if (state === "no_jwt") return T("bal_no_jwt");
  if (state === "auth") return T("bal_none");
  // No state at all means an older engine answered, or the account carries no
  // JWT to read with. Both keep the original re-login line rather than
  // inventing a cause we cannot actually see.
  return T("bal_none");
}

/** Per-account balance list, honouring the search box + filter dropdown. */
function renderBalanceList() {

    if (f === "all") return true;
    if (f === "zai" || f === "bigmodel") return a.provider === f;
    return (meta && meta.plan === f) || a.plan === f;
  });
  if (!rows.length) {
    el.innerHTML = '<div class="bal-empty">' + (lastQuota.length ? T("no_match") : T("bal_none")) + "</div>";
    return;
  }

    const meta = lastAccountMeta.get(a.id);
    const box = document.createElement("div");
    // Model-scoped holds, same rule as the accounts page: a held model means the
    // account is degraded, not down. The overview showed only the main status,
    // so a throttled account looked fully healthy here even while requests to
    // the held model were failing.
    const holds = (meta && meta.modelCooldowns) || [];
    const throttleBadge = holds.length
      ? '<span class="st throttled">' + T("st_throttled") + " · " + holds.length + "</span>"
      : "";
    // The status class drives the card's accent bar, so an unhealthy account is
    // distinguishable at a glance rather than only by reading its badge.
    const accent = meta && meta.status === "ok" && holds.length ? "throttled" : (meta ? meta.status : "");
    box.className = "bal-acc" + (accent ? " " + accent : "");
    const status = meta ? '<span class="st ' + meta.status + '">' + T("st_" + meta.status) + "</span>" + throttleBadge : "";
    const planPill = '<span class="pill ' + ((meta && meta.plan === "start-plan") ? "plan-free" : "plan-paid") + '">' +
      T((meta && meta.plan === "start-plan") ? "plan_free" : "plan_paid") + "</span>";
    /**
     * Claim button state.
     *
     * "Nothing left to claim" and "nothing was ever offered" are different
     * things, and the button alone cannot tell them apart — which is why it
     * used to sit there enabled forever, inviting a click that always failed
     * with "today's already claimed". The distinction is already in the
     * payload: `claimablePlans` is what upstream is still OFFERING, so an empty
     * list alongside an active plan means the offer was taken. Only the second
     * case is labelled 已领取; a brand-new account with no plan at all keeps the
     * button, because for it the claim is still ahead.
     */
    const claimable = (a.claimablePlans || []).length;
    const active = (a.activePlans || []).length;
    const alreadyClaimed = claimable === 0 && active > 0;
    const claimBtn = alreadyClaimed
      ? '<span class="pill claim-done" title="' + T("claim_done_hint") + '">' + T("claim_done") + "</span>"
      : '<button class="btn sm ghost claim-one" data-acc="' + esc(a.id) + '">' + T("claim_one") + "</button>";
    box.innerHTML = '<div class="bal-acc-head"><span class="nm">' + esc(a.name) + "</span>" +
      '<span class="pill ' + esc(a.provider) + '">' + esc(a.provider) + "</span>" + planPill + status +
      claimBtn +
      "</div>";
    if (!(a.balances || []).length) {
      const none = document.createElement("div");
      none.className = "bal-empty";
      none.textContent = balanceEmptyText(a);
      box.appendChild(none);
    } else {
      for (const b of a.balances) {
        const pct = b.totalUnits > 0 ? Math.max(0, Math.min(100, b.remainingUnits / b.totalUnits * 100)) : 0;
        const row = document.createElement("div");
        row.className = "bal-row";
        // The grant is a daily allowance: the number that matters is not just
        // "how much is left" but "how long until it is gone either way". Upstream
        // reports `expiresAt` in SECONDS (a 10-digit value), so it must be
        // multiplied before use — reading it as milliseconds dates it to 1970.
        // Absent for an account whose bucket has no expiry, in which case the
        // countdown is simply not rendered rather than showing a fake one.
        const exp = Number(b.expiresAt);
        const expiry = Number.isFinite(exp) && exp > 0
          ? '<div class="bal-exp">' + T("bal_expires") + " " +
            '<span class="cd" data-until="' + (exp * 1000) + '"></span></div>'
          : "";
        row.innerHTML = '<div class="bal-top"><span>' + esc(b.showName) + "</span><span>" +
          fmtNum(b.remainingUnits) + " / " + fmtNum(b.totalUnits) + "</span></div>" +
          '<div class="bar' + (pct < 20 ? " low" : "") + '"><i style="width:' + pct + '%"></i></div>' +
          expiry;
        box.appendChild(row);
      }
    }
    // Grants already on the account but not yet usable. Shown separately from
    // balances because a pending activity package contributes ZERO to every
    // bucket until its effective_at — without this line the panel looks like
    // the claim failed.
    for (const p of (a.activePlans || [])) {
      if (!p.pending) continue;
      const note = document.createElement("div");
      note.className = "pending-grant";
      const total = (p.entitlements || []).reduce((s, e) => s + (e.grantUnits || 0), 0);
      const when = p.effectiveAt ? fmtLocalTime(p.effectiveAt) : "—";
      note.innerHTML = '<span class="pg-tag">' + T("pg_pending") + "</span>" +
        '<span class="pg-txt">' + esc(p.name) + " · " + fmtNum(total) + " " + T("pg_units") + "</span>" +
        '<span class="pg-when">' + T("pg_effective") + " " + esc(when) + "</span>";
      box.appendChild(note);
    }
    /**
     * Countdown to when the plan's daily allowance STOPS.
     *
     * A free account's daily grant is not permanent: upstream reports an `endsAt`
     * on the plan, and once it passes the free allowance is gone (measured: an
     * account registered 09-22 carries endsAt 09-26 23:59:59 — four days). The
     * balances above keep showing a number until then, so without this line the
     * drop to zero looks like it came from nowhere. An activity package has its
     * own, usually much shorter, window, which is why each plan gets a line.
     */
    for (const p of (a.activePlans || [])) {
      if (p.pending || !p.endsAt) continue;
      const line = document.createElement("div");
      line.className = "plan-exp";
      const total = (p.entitlements || []).reduce((s, e) => s + (e.grantUnits || 0), 0);
      line.innerHTML = '<span class="pe-name">' + esc(p.name) + "</span>" +
        (total ? '<span class="pe-units">' + fmtNum(total) + "</span>" : "") +
        '<span class="pe-tag">' + T("pg_ends") + "</span>" +
        '<span class="cd" data-until="' + (p.endsAt * 1000) + '" data-fmt="days"></span>';
      box.appendChild(line);
    }
    el.appendChild(box);
  }
}

/**
 * Render a unix-seconds timestamp in the viewer's local time.
 *
 * The server reports UTC seconds; showing them raw would make a 23:00 local
 * activation read as 15:00 and look wrong. Local time is what the user plans
 * around, so convert here and keep the date when it is not today.
 */
function fmtLocalTime(sec) {
  const d = new Date(sec * 1000);
  const now = new Date();
  const hhmm = String(d.getHours()).padStart(2, "0") + ":" + String(d.getMinutes()).padStart(2, "0");
  const sameDay = d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth() && d.getDate() === now.getDate();
  return sameDay ? hhmm : (d.getMonth() + 1) + "/" + d.getDate() + " " + hhmm;
}

/** Claim for one account (or all). Reports each account's own outcome. */
async function doClaim(payload, btn) {
  if (btn) btn.disabled = true;
  try {
    const r = await api("/accounts/claim", { method: "POST", body: JSON.stringify(payload) });
    const d = await r.json();
    const results = d.results || [];
    if (!results.length) { toast(T("claim_none")); return; }
    const okCount = results.filter(x => x.ok).length;
    if (okCount > 0) {
      toast(T("claim_ok").replace("{n}", String(okCount)));
    } else {
      // Surface the server's own reason — "nothing to claim" and "captcha
      // failed" need very different reactions from the user.
      const first = results[0];
      const reasonKey = "claim_r_" + (first.reason || "error");
      const known = T(reasonKey) !== reasonKey;
      let reasonText = known ? T(reasonKey) : (first.message || first.reason || "");
      // Append upstream's own text when it says something the fixed label does
      // not. The label for `ineligible` is "this account is not eligible", which
      // is true but useless — the operator's next question is always WHY, and
      // the gateway usually answers it ("needs app version >= X", "plan already
      // held", a region or campaign window). The label alone threw that away.
      const detail = (first.message || "").trim();
      if (known && detail && !reasonText.includes(detail)) reasonText += "（" + detail + "）";
      toast(T("claim_fail") + " " + reasonText);
    }
    loadBalances();
  } catch (e) { toast(String(e)); }
  finally { if (btn) btn.disabled = false; }
}

let lastAccounts = [];
/**
 * Refresh the accounts page AND the overview's balance list.
 *
 * The overview renders its own copy of the pool (ov-balances, built from
 * loadBalances) and nothing tied the two together, so deleting an account here
 * left it on the overview until the operator reloaded the page. Both views read
 * the same endpoints, so refreshing them together costs nothing extra and is
 * what "the pool changed" should always mean.
 */
async function refreshAccounts() {
  const r = await api("/accounts");
  const j = await r.json();
  lastAccounts = j.accounts || [];
  renderAccounts(lastAccounts);
  renderStoreProblem(j.storeProblem);
  // ONE quota read, shared by both views. Each used to fetch it on its own, so a
  // single refresh sent two identical /accounts/quota requests — and for an
  // account the poller has no snapshot for, that endpoint performs a LIVE
  // billing read rather than answering from cache, so the duplicate was a real
  // upstream burst doubled, not just a local round trip.
  let quota = null;
  try { quota = await api("/accounts/quota").then((x) => x.json()); }
  catch (e) { /* both consumers fall back to their own handling of "no data" */ }
  fillAccountBalances(quota);
  // The overview's own list, kept in sync so a deletion shows up immediately.
  await refreshOverviewBalances(quota);
}

/**
 * Warn when the account store exists but could not be read.
 *
 * An empty account list has two very different causes and they look identical
 * on screen: nothing was ever configured, or the store is there but encrypted
 * with a key this host cannot derive — which is what an import from another
 * machine produces (Docker especially, since the container's home and platform
 * differ from the desktop's). The second case needs a re-login, and without this
 * banner the operator just sees their accounts missing and no explanation.
 */
function renderStoreProblem(problem) {
  const el = document.getElementById("acc-store-problem");
  if (!el) return;
  if (!problem) { el.style.display = "none"; el.innerHTML = ""; return; }
  const key = problem.kind === "undecryptable_locked" ? "store_problem_locked" : "store_problem";
  el.innerHTML = T(key).replace("{f}", esc(problem.file || ""));
  el.style.display = "";
}

/**
 * Re-render the overview balance list from the cached quota endpoint.
 *
 * Deliberately reuses loadBalances rather than patching lastQuota in place: the
 * quota payload carries per-account rows the accounts endpoint does not (and
 * vice versa), so filtering one list locally would leave the totals chips stale.
 */
async function refreshOverviewBalances(knownQuota) {
  // Only when the overview has been rendered at least once — otherwise this
  // would fire a needless quota request for a page the operator has not opened.
  if (!document.getElementById("ov-balances").dataset.loaded) return;
  // Reuses the pool AND the quota payload refreshAccounts just fetched, so this
  // costs no extra requests and cannot show a pool older than the accounts page.
  await loadBalances(lastAccounts, knownQuota);
}

/** Apply the account-page search box + filter to the cached list. */
function applyAccountFilter() {
  const q = (document.getElementById("acc-search").value || "").trim().toLowerCase();
  const f = document.getElementById("acc-filter").value;
  const list = lastAccounts.filter(a => {
    if (q && !(a.name || "").toLowerCase().includes(q)) return false;
    if (f === "all") return true;
    if (f === "zai" || f === "bigmodel") return a.provider === f;
    if (f === "coding-plan" || f === "start-plan") return a.plan === f;
    return a.status === f;
  });
  renderAccounts(list);
  const cnt = document.getElementById("acc-count");
  if (cnt) cnt.textContent = list.length === lastAccounts.length
    ? lastAccounts.length + T("acc_unit")
    : list.length + " / " + lastAccounts.length + T("acc_unit");
  void fillAccountBalances();
}

/**
 * Fill the per-card balance bars.
 *
 * `knownQuota` is the already-fetched /accounts/quota payload, when the caller
 * has one. Both this and the overview list used to fetch that endpoint
 * independently, so one pool refresh fired two identical requests — and for an
 * account the poller has no snapshot for, that endpoint performs a LIVE billing
 * read instead of answering from cache, which made the duplicate double a real
 * upstream burst rather than a local round trip.
 */
async function fillAccountBalances(knownQuota) {
  let data;
  if (knownQuota) {
    data = knownQuota;
  } else {
    try {
      data = await (await api("/accounts/quota")).json();
    } catch { return; }
  }
  const byId = new Map((data.accounts || []).map(a => [a.id, a]));
  document.querySelectorAll(".acc-bal").forEach(el => {
    const rec = byId.get(el.dataset.acc);
    el.innerHTML = "";
    if (!rec || !(rec.balances || []).length) {
      el.innerHTML = '<div class="bal-none">' + esc(balanceEmptyText(rec)) + "</div>";
      return;
    }
    for (const b of rec.balances) {
      const pct = b.totalUnits > 0 ? Math.max(0, Math.min(100, b.remainingUnits / b.totalUnits * 100)) : 0;
      const row = document.createElement("div");
      row.className = "bal-row";
      // Same seconds→ms guard as the overview list: upstream reports `expiresAt`
      // in seconds, so it must be multiplied before use.
      const exp = Number(b.expiresAt);
      const expiry = Number.isFinite(exp) && exp > 0
        ? '<div class="bal-exp">' + T("bal_expires") + " " +
          '<span class="cd" data-until="' + (exp * 1000) + '"></span></div>'
        : "";
      row.innerHTML = '<div class="bal-top"><span>' + esc(b.showName) + "</span><span>" +
        fmtNum(b.remainingUnits) + " / " + fmtNum(b.totalUnits) + "</span></div>" +
        '<div class="bar' + (pct < 20 ? " low" : "") + '"><i style="width:' + pct + '%"></i></div>' +
        expiry;
      el.appendChild(row);
    }
    // Plan expiry, so the pool page answers "how long does this account's free
    // allowance last" without switching to the overview. Same rule as there:
    // only a live (non-pending) plan with an `endsAt` gets a countdown.
    for (const p of (rec.activePlans || [])) {
      if (p.pending || !p.endsAt) continue;
      const line = document.createElement("div");
      line.className = "plan-exp";
      line.innerHTML = '<span class="pe-name">' + esc(p.name) + "</span>" +
        '<span class="pe-tag">' + T("pg_ends") + "</span>" +
        '<span class="cd" data-until="' + (p.endsAt * 1000) + '" data-fmt="days"></span>';
      el.appendChild(line);
    }
    // The claim state belongs here too: this is the page where an operator
    // decides which account to use, and "already claimed today" is part of that.
    const claimed = (rec.claimablePlans || []).length === 0 && (rec.activePlans || []).length > 0;
    if (claimed) {
      const done = document.createElement("div");
      done.className = "bal-none claim-done-line";
      done.textContent = T("claim_done");
      el.appendChild(done);
    }
  });
}
function renderAccounts(list) {
  const el = document.getElementById("acc-list");
  el.innerHTML = "";
  if (!list.length) {
    el.innerHTML = '<div class="card empty" style="grid-column:1/-1"><img class="big" src="/admin/api/logo.svg" width="46" height="46" alt=""><div>' +
      T("no_accounts") + '</div><div class="sub">' + T("no_accounts_sub") + "</div></div>";
    return;
  }
  for (const a of list) {
    const card = document.createElement("div");
    // Model-scoped holds. Upstream throttles PER MODEL, so this account is still
    // serving everything else — it gets its own badge rather than replacing the
    // status, which would claim the whole account is down. Without it a
    // throttled account showed a green "正常" while every request to the held
    // model failed, which is what the user saw during the 405/3012 bursts.
    const holds = a.modelCooldowns || [];
    const throttleBadge = holds.length
      ? '<span class="st throttled">' + T("st_throttled") + " · " + holds.length + "</span>"
      : "";
    // The accent bar follows the badge, and picks up `throttled` when that is
    // the only thing wrong: a held model is a degraded account, and the card
    // should not look identical to a fully healthy one.
    const accent = a.status === "ok" && holds.length ? "throttled" : a.status;
    card.className = "card acc-card " + accent;
    const holdLines = holds.map((h) =>
      '<div class="hold-line">' + esc(h.model) + " — " + T("hold_retry_in") + " " +
      '<span class="cd" data-until="' + (Date.now() + h.remainingMs) + '"></span></div>').join("");
    card.innerHTML =
      '<div class="acc-head"><span class="acc-name">' + esc(a.name) + "</span>" +
      '<span class="pill ' + a.provider + '">' + a.provider + "</span>" +
      '<span class="pill ' + (a.plan === "start-plan" ? "plan-free" : "plan-paid") + '">' +
        (a.plan === "start-plan" ? T("plan_free") : T("plan_paid")) + "</span>" +
      '<span class="st ' + a.status + '">' + T("st_" + a.status) + "</span>" + throttleBadge +
      (a.isDefault ? '<span class="pill plan-paid">' + T("act_default") + "</span>" : "") + "</div>" +
      (a.statusNote ? '<div class="sub" style="margin:6px 0 0">' + esc(a.statusNote) + "</div>" : "") +
      (holdLines ? '<div class="hold-box">' + holdLines + "</div>" : "") +
      '<div class="meta-row"><span>' + T("concurrency") + " " + a.inFlight + "/" + a.maxConcurrent + "</span>" +
      "<span>" + T("reqs") + " " + a.stats.requests + "</span>" +
      "<span>" + T("fail") + " " + a.stats.failures + "</span>" +
      "<span>" + T("th_cool") + " " + a.stats.cooldowns + "</span>" +
      "<span>mid…" + esc(a.deviceMidTail) + "</span>" +
      (a.cooldownRemainingMs > 0 ? "<span style='color:var(--gold)'>" + Math.ceil(a.cooldownRemainingMs / 1000) + "s</span>" : "") + "</div>" +
      // Balance bars are filled asynchronously from /accounts/quota (cached
      // per refresh so the card list itself stays a single request).
      '<div class="acc-bal" data-acc="' + a.id + '"></div>';
    const actions = document.createElement("div");
    actions.className = "acc-actions";
    const mkBtn = (label, cls, fn, blockedReason) => {
      const b = document.createElement("button");
      b.className = "btn sm " + cls; b.textContent = label; b.onclick = fn;
      // Present but inert, with the reason on hover. Removing a control the
      // operator is used to reads as "someone deleted an option"; showing it
      // enabled when the click cannot stick is worse. Disabled says both.
      if (blockedReason) { b.disabled = true; b.title = blockedReason; }
      actions.appendChild(b);
    };
    if (a.status === "paused") {
      mkBtn(T("act_resume"), "ghost", () => act("/accounts/pause", a.id, false));
    } else {
      mkBtn(T("act_pause"), "ghost", () => act("/accounts/pause", a.id, true));
    }
    if (a.status === "cooldown" || a.status === "exhausted") {
      mkBtn(T("act_clear"), "gold", () => act("/accounts/resume", a.id));
    }
    // Plan tier: always shown, disabled where it cannot take effect.
    //
    // The tier is DERIVED from the credential on every load — a credential with
    // a JWT is start-plan, one without is coding-plan (see repairPlanTier) — and
    // upstream refuses a JWT credential on the coding-plan endpoints, so the
    // derivation is not something the panel can override. On a JWT account the
    // button used to flip its own label and then flip back on the next reload,
    // with nothing to explain why. It was hidden for a while, which read as "an
    // option went missing"; disabled with the reason on hover is the honest and
    // less alarming version.
    mkBtn(a.plan === "start-plan" ? T("act_paid") : T("act_free"), "ghost",
      () => act("/accounts/update", a.id, undefined, { plan: a.plan === "start-plan" ? "coding-plan" : "start-plan" }),
      a.hasJwt ? T("plan_locked_jwt") : "");
    // Re-run the OAuth flow for an existing account: the direct way to refresh a
    // credential upstream has started rejecting, without deleting the account
    // (which would drop its deviceMid and request history). Highlighted when the
    // account is actually in that state — it is the action that resolves it, and
    // a dead account is otherwise indistinguishable from a healthy one at a glance.
    mkBtn(T("act_relogin"), a.status === "relogin" ? "gold" : "ghost", () => {
      document.getElementById("login-mask").classList.add("on");
      document.getElementById("login-wait").style.display = "none";
      document.getElementById("login-provider").value = a.provider;
      document.getElementById("login-name").value = a.name;
      renderExistingInModal();
    });
    if (!a.isDefault) {
      mkBtn(T("act_default"), "ghost", async () => {
        const r = await api("/accounts/default", { method: "POST", body: JSON.stringify({ id: a.id }) });
        if (!(await okOrToast(r))) return;
        refreshAccounts();
      });
    }
    mkBtn(T("act_delete"), "danger", async () => {
      // Destructive and irreversible from the panel (the credential is gone),
      // so it gets the danger styling and the delete verb on the button.
      if (!(await askConfirm(T("confirm_delete"), { danger: true, okText: T("act_delete") }))) return;
      const r = await api("/accounts/delete", { method: "POST", body: JSON.stringify({ id: a.id }) });
      // A silent failure here is the worst case in the panel: the operator
      // confirmed a destructive action, nothing was deleted, and the card still
      // sitting there is the only clue.
      if (!(await okOrToast(r))) return;
      refreshAccounts();
    });
    card.appendChild(actions);
    el.appendChild(card);
  }
}
/**
 * Report a failed panel call. True when it succeeded.
 *
 * Every account button used to fire its request and refresh, ignoring the
 * answer — so a REJECTED operation looked exactly like a successful one: the
 * card simply did not change, and the operator clicked again. Delete was the
 * worst of them, because the confirm dialog made it look like something had
 * happened. The store refusing to persist a pause is the same shape: the change
 * applies in memory and is gone after the next restart, with nothing to say so.
 */
async function okOrToast(resp) {
  if (resp.ok) return true;
  let msg = "HTTP " + resp.status;
  try {
    const j = await resp.json();
    if (j && j.error && j.error.message) msg = j.error.message;
  } catch (e) { /* non-JSON body: the status is the message */ }
  toast(msg);
  return false;
}

async function act(path, id, paused, extra) {
  const r = await api(path, { method: "POST", body: JSON.stringify({ id, ...(paused !== undefined ? { paused } : {}), ...(extra || {}) }) });
  // Refresh only on success. Repainting the unchanged card after a failure is
  // what made a rejected click indistinguishable from a dead button.
  if (!(await okOrToast(r))) return;
  refreshAccounts();
}

  document.getElementById("login-start").click();
};
/**
 * Copy text to the clipboard with a non-secure-context fallback.
 *
 * `navigator.clipboard` exists only in secure contexts: https, or http on
 * 127.0.0.1 which browsers whitelist. The panel reached over a LAN IP is
 * plain http, where `navigator.clipboard` is undefined — the direct calls
 * threw and every copy button silently died. The textarea + execCommand path
 * still works there, so every copy goes through this helper.
 */
async function copyToClipboard(text) {
  try {
    await navigator.clipboard.writeText(text);
    return true;
  } catch {
    // Undefined (non-secure context) or denied — fall through.
  }
  try {
    const ta = document.createElement("textarea");
    ta.value = text;
    ta.style.position = "fixed";
    ta.style.opacity = "0";
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand("copy");
    document.body.removeChild(ta);
    return ok;
  } catch {
    return false;
  }
}
/**
 * Copy the authorize link.
 *
 * `navigator.clipboard` needs a secure context; the panel is plain HTTP on
 * 127.0.0.1, which browsers do treat as secure, but a LAN address does not. The
 * textarea fallback keeps the button working there instead of failing silently,
 * and the link is shown in the toast either way so it can always be selected by
 * hand.
 */
document.getElementById("login-copy").onclick = async () => {
  const url = document.getElementById("login-url").textContent.trim();
  if (!url) { toast(T("picker_none")); return; }
  toast((await copyToClipboard(url)) ? T("copied") : T("copy_manual"));
};
/**
 * Open the authorize page in a local private window.
 *
 * The engine launches Edge (`--inprivate`) or Chrome (`--incognito`) because a
 * page cannot ask for a private window and `window.open` is popup-blocked here.
 * The link is also always visible and copyable, so a failure is never a dead
 * end — the toast says what to do instead of silently doing nothing.
 */
document.getElementById("login-open").onclick = async (e) => {
  const url = document.getElementById("login-url").textContent.trim();
  if (!url) { toast(T("picker_none")); return; }
  const btn = e.target;
  btn.disabled = true;
  try {
    const r = await api("/open-url", { method: "POST", body: JSON.stringify({ url }) });
    const d = await r.json();
    if (d.ok) {
      toast(T("open_ok").replace("{b}", d.browser === "edge" ? "Edge" : "Chrome"));
    } else {
      // Fall back to a plain tab so the user is not stuck; the private window is
      // a preference, not a requirement for authorizing.
      toast(d.message || T("popup_blocked"));
      window.open(url, "_blank", "noopener,noreferrer");
    }
  } catch (err) {
    toast(T("popup_blocked"));
    window.open(url, "_blank", "noopener,noreferrer");
  }
  btn.disabled = false;
};

/**
 * Check the authorization result now.
 *
 * The background poll runs every 3s, but the user has just come back from
 * another device and does not know that — waiting up to 3s with no feedback
 * reads as "nothing happened". This asks immediately and reports what it found,
 * so the button either completes the flow or confirms it is still pending.
 */
document.getElementById("login-recheck").onclick = async (e) => {
  const btn = e.target;
  const sid = document.getElementById("login-wait").dataset.sid;
  if (!sid) { toast(T("picker_none")); return; }
  btn.disabled = true;
  const hint = document.getElementById("login-hint");
  const prev = hint.textContent;
  hint.textContent = T("login_checking");
  try {
    const r = await api("/accounts/login/status", { method: "POST", body: JSON.stringify({ sessionId: sid }) });
    const d = await r.json();
    if (d.done && d.ok) {
      // Same completion path as the poll, so both routes behave identically.
      stopCountdown();
      toast(T("login_ok"));
      document.getElementById("login-mask").classList.remove("on");
      refreshAccounts(); refreshOverview(); loadBalances();
      return;
    }
    if (d.done) { toast(T("login_fail") + (d.error || "")); return; }
    if (d.expired) {
      stopCountdown();
      document.getElementById("login-spin").style.display = "none";
      document.getElementById("login-hint").style.display = "none";
      document.getElementById("login-expired").style.display = "block";
      return;
    }
    toast(T("login_pending"));
  } catch (err) {
    toast(T("login_fail") + String(err));
  } finally {
    if (hint.textContent === T("login_checking")) hint.textContent = prev;
    btn.disabled = false;
  }
};

let cdTimer = null;
function stopCountdown() {
  if (cdTimer !== null) { clearInterval(cdTimer); cdTimer = null; }
}
/**
 * Render the remaining validity of the authorize link as mm:ss.
 *
 * The link is single-use and short-lived: once it expires the upstream flow is
 * dead and the panel cannot revive it, so the countdown turns amber in the last
 * minute and, at zero, swaps the spinner for an explicit "regenerate" action
 * instead of leaving the user staring at a spinner that will never resolve.
 */
function startCountdown(deadline) {
  stopCountdown();
  const el = document.getElementById("login-countdown");
  const tick = () => {
    const left = Math.max(0, deadline - Date.now());
    const s = Math.floor(left / 1000);
    el.textContent = " · " + Math.floor(s / 60) + ":" + String(s % 60).padStart(2, "0");
    el.classList.toggle("cd-warn", left <= 60_000 && left > 0);
    el.classList.toggle("cd-dead", left <= 0);
    if (left <= 0) {
      stopCountdown();
      document.getElementById("login-spin").style.display = "none";
      document.getElementById("login-hint").style.display = "none";
      document.getElementById("login-expired").style.display = "block";
    }
  };
  tick();
  cdTimer = setInterval(tick, 1000);
}
async function pollLogin(sid) {
  const wait = document.getElementById("login-wait");
  if (wait.dataset.sid !== sid) return;
  try {
    const r = await api("/accounts/login/status", { method: "POST", body: JSON.stringify({ sessionId: sid }) });
    const d = await r.json();
    if (d.done) {
      stopCountdown();
      if (d.ok) {
        toast(T("login_ok"));
        document.getElementById("login-mask").classList.remove("on");
        refreshAccounts(); refreshOverview(); loadBalances();
      } else { toast(T("login_fail") + (d.error || "")); }
      return;
    }
    // The server is authoritative about expiry: if it says the session is gone,
    // stop polling rather than hammering a dead id forever.
    if (d.expired) {
      stopCountdown();
      document.getElementById("login-spin").style.display = "none";
      document.getElementById("login-hint").style.display = "none";
      document.getElementById("login-expired").style.display = "block";
      return;
    }
  } catch (e) {
    // 404 unknown_session — the server already dropped it (TTL elapsed).
    if (String(e).includes("unknown_session") || String(e).includes("404")) {
      stopCountdown();
      document.getElementById("login-spin").style.display = "none";
      document.getElementById("login-hint").style.display = "none";
      document.getElementById("login-expired").style.display = "block";
      return;
    }
  }
  setTimeout(() => pollLogin(sid), 3000);
}

/**
 * Playground conversation, kept across reloads.
 *
 * It used to live only in memory, so a refresh — or a page switch and back —
 * silently discarded a conversation that may have taken several turns to build.
 * For a page whose whole job is "verify the chain end to end", losing the
 * evidence on refresh is the wrong default.
 *
 * Entries are `user` | `assistant` | `error`. The error rows are part of the
 * SAVED conversation on purpose: when a send fails, the failure text is the
 * evidence of what happened, and it is the thing the operator most wants to find
 * still there after a refresh. They are filtered out of the payload sent to the
 * model (see pgSend) — the upstream would reject a message with role "error".
 *
 * Stored on the engine (`data/playground.json`), with a localStorage copy kept
 * only as a fallback for an unreachable engine. It lived in localStorage alone
 * before, which meant clearing the engine's data directory did not clear it —
 * a fresh install still showed a full conversation — and the same operator
 * opening the panel from another browser saw nothing.
 */
/** The model the transcript's most recent entries were produced with. */
let pgLastModel = "";

/**
 * The picker selections, kept alongside the conversation.
 *
 * Neither the model nor the account was remembered, so every reload re-seeded
 * the model to the FIRST entry of the list — pick `glm-5.3-flash`, refresh, and
 * you are back on `glm-5.3`. Worse, the conversation itself IS restored, so the
 * transcript came back with a selector that disagreed with the replies already
 * in it: the next message went to a different model than the one that produced
 * everything above it, and nothing on screen said so.
 */
const PG_SEL = "zk-pg-select";

const PG_STORE = "zk-pg-history";
let pgHistory = [];
/** Restored once per page load, so re-entering the page cannot duplicate it. */
let pgRestored = false;
/** Pending debounced upload, so a clear can cancel one that has not fired. */
let pgPushTimer = null;

/**
 * Persist the transcript to the engine, and keep a browser copy as a fallback.
 *
 * The engine owns this now (`data/playground.json`). It used to live only in
 * localStorage, which put it beyond the reach of the one action that clears
 * state: wiping the engine's data directory left the conversation sitting in
 * the browser, so a fresh install still showed a full transcript. It also made
 * the transcript invisible to the engine itself, so opening the panel from
 * another browser showed an empty page right after running a test.
 *
 * localStorage is still written, but as a fallback rather than the source of
 * truth: it is what the next page load falls back to when the engine cannot be
 * reached. Neither write may break sending, so both are guarded.
 */
function pgSave() {
  try { localStorage.setItem(PG_STORE, JSON.stringify(pgHistory)); }
  catch (e) { /* private mode, or quota — losing the copy must not break sending */ }
  pgSaveSelection();
  pgPushToServer();
}

/**
 * Upload the transcript, coalescing rapid saves.
 *
 * pgSave runs on every message AND on every picker change, and each call sends
 * the whole transcript — without the debounce a streaming reply would upload
 * the conversation repeatedly as it grows.
 */
function pgPushToServer() {
  if (pgPushTimer) clearTimeout(pgPushTimer);
  pgPushTimer = setTimeout(() => {
    pgPushTimer = null;
    const sel = pgLoadSelection();
    api("/playground/history", {
      method: "POST",
      body: JSON.stringify({
        entries: pgHistory,
        selection: { model: sel.model, account: sel.account },
      }),
    }).catch(() => { /* engine unreachable — the localStorage copy still has it */ });
  }, 400);
}

/** The browser's fallback copy, filtered to entries the panel can repaint. */
function pgSavedLocally() {
  try {
    const raw = localStorage.getItem(PG_STORE);
    const v = raw ? JSON.parse(raw) : null;
    return Array.isArray(v) ? v : [];
  } catch (e) { return []; }
}

/**
 * Persist the picker choices, so a reload does not silently change the model.
 *
 * Defensive on purpose: it is called from pgSave (the conversation is the thing
 * that must never fail to save) and reads two DOM nodes that may not exist in a
 * given context. A missing element or a locked-down localStorage must not take
 * the conversation down with it.
 */
function pgSaveSelection() {
  try {
    const model = (typeof document !== "undefined" && document.getElementById("pg-model"))?.value ?? "";
    const account = (typeof document !== "undefined" && document.getElementById("pg-account"))?.value ?? "";
    localStorage.setItem(PG_SEL, JSON.stringify({ model, account }));
  } catch (e) { /* private mode: the session simply does not survive a reload */ }
}

/** The stored selection, or nulls. Never throws. */
function pgLoadSelection() {
  try {
    const raw = localStorage.getItem(PG_SEL);
    const v = raw ? JSON.parse(raw) : null;
    return v && typeof v === "object" ? v : { model: "", account: "" };
  } catch (e) {
    return { model: "", account: "" };
  }
}

/** Repaint a stored conversation. No-op once restored, or when there is none. */
function pgRestore() {
  if (pgRestored) return;
  pgRestored = true;
  // The engine is asked first: its copy is the one that survives a browser
  // profile reset and the one that clearing data/ removes, so it is the source
  // of truth. The browser copy is the fallback, for a transcript written before
  // this moved server-side and for an unreachable engine.
  api("/playground/history")
    .then((r) => r.json())
    .then((d) => {
      const entries = Array.isArray(d && d.entries) ? d.entries : [];
      if (entries.length) { pgPaint(entries); return; }
      const local = pgSavedLocally();
      if (local.length) { pgPaint(local); pgPushToServer(); }
    })
    .catch(() => {
      const local = pgSavedLocally();
      if (local.length) pgPaint(local);
    });
}

/**
 * Repaint `list` into the transcript.
 *
 * Filtered rather than trusted: an entry arrives from the engine's JSON file or
 * from localStorage, both editable, and a bad role would be replayed upstream
 * as a malformed message.
 */
function pgPaint(list) {
  pgHistory = list.filter((m) => m && typeof m.content === "string" && PG_ROLES.has(m.role));
  for (const m of pgHistory) chatBubble(pgBubbleClass(m.role), m.content);
}
/** The roles a saved entry may carry. `note` is a transcript divider. */
const PG_ROLES = new Set(["user", "assistant", "error", "note"]);
/** Which bubble class a stored role repaints as. */
function pgBubbleClass(role) {
  return role === "user" ? "user" : role === "error" ? "err" : role === "note" ? "note" : "bot";
}

/**
 * Note in the transcript that the model changed.
 *
 * The reply after this point came from a different model, and a transcript that
 * does not say so is misleading: comparing two answers is the main reason to
 * switch mid-conversation, and the comparison needs to be readable afterwards.
 * Recorded in the history (role `note`) so it survives a refresh with the rest
 * of the conversation; it is filtered out of the payload sent upstream, which
 * accepts only user/assistant.
 */
function pgNoteSwitch(from, to) {
  if (!from || !to || from === to) return;
  const text = T("pg_switched").replace("{from}", from).replace("{to}", to);
  pgHistory.push({ role: "note", content: text });
  pgSave();
  chatBubble("note", text);
}

/**
 * Searchable picker — a modal replacement for native <select>.
 *
 * Native selects are unusable here for two reasons: the list scrolls with no
 * way to filter, and each row can only be a bare string, so account status and
 * provider badges cannot be shown. This keeps the same "one value + one label"
 * contract while adding a search box, keyboard navigation and rich rows.
 *
 * Usage: a `<button class="picker-btn" data-picker="<hiddenInputId>">` plus a
 * hidden input with that id. Register options with `setPickerOptions(id, [...])`;
 * the hidden input's `value` is the selection and its `onchange` fires on pick.
 */
const pickerState = {
  /** id -> { options, filter } where options are {value,label,sub,badges} */
  registry: {},
  openId: null,
  activeIndex: 0,
  filtered: [],
};

/** Define (or replace) the options for a picker. */
function setPickerOptions(id, options) {
  pickerState.registry[id] = options;
  // Keep the button's caption in sync with the current selection.
  const hidden = document.getElementById(id);
  const btn = document.querySelector(`.picker-btn[data-picker="${id}"]`);
  if (!hidden || !btn) return;
  const sel = options.find(o => o.value === hidden.value);
  btn.innerHTML = "";
  const span = document.createElement("span");
  span.className = "pi-name";
  span.textContent = sel ? sel.label : (options[0] ? options[0].label : T("picker_none"));
  btn.appendChild(span);
  if (sel && sel.badges) {
    for (const b of sel.badges) {
      const s = document.createElement("span");
      s.className = "st " + (b.cls || "");
      s.textContent = b.text;
      btn.appendChild(s);
    }
  }
}

/** Programmatically choose a value (also updates the button caption). */
function setPickerValue(id, value) {
  const hidden = document.getElementById(id);
  if (!hidden) return;
  hidden.value = value;
  setPickerOptions(id, pickerState.registry[id] || []);
}

function openPicker(id) {
  const opts = pickerState.registry[id] || [];
  if (!opts.length) return;
  pickerState.openId = id;
  pickerState.activeIndex = 0;
  const titleKey = document.querySelector(`.picker-btn[data-picker="${id}"]`)?.dataset.title;
  document.getElementById("picker-title").textContent = titleKey ? T(titleKey) : T("picker_title");
  const search = document.getElementById("picker-search");
  search.value = "";
  renderPickerList("");
  document.getElementById("picker-mask").classList.add("on");
  setTimeout(() => search.focus(), 40);
}

function closePicker() {
  pickerState.openId = null;
  document.getElementById("picker-mask").classList.remove("on");
}

/** Render the filtered list; called on open and on every keystroke. */
function renderPickerList(query) {
  const id = pickerState.openId;
  if (!id) return;
  const q = (query || "").trim().toLowerCase();
  const all = pickerState.registry[id] || [];
  // Match against every field the user can see, so typing an IP-like tail or a
  // provider name both work.
  const list = q
    ? all.filter(o => `${o.label} ${o.sub || ""} ${o.value}`.toLowerCase().includes(q))
    : all;
  pickerState.filtered = list;
  if (pickerState.activeIndex >= list.length) pickerState.activeIndex = 0;

  const box = document.getElementById("picker-list");
  box.innerHTML = "";
  if (!list.length) {
    box.innerHTML = '<div class="picker-empty">' + T("picker_no_match") + "</div>";
    return;
  }
  const cur = document.getElementById(id)?.value;
  list.forEach((o, i) => {
    const row = document.createElement("div");
    row.className = "picker-item" + (i === pickerState.activeIndex ? " active" : "");
    const name = document.createElement("span");
    name.className = "pi-name";
    name.textContent = o.label;
    row.appendChild(name);
    if (o.sub) {
      const sub = document.createElement("span");
      sub.className = "pi-sub";
      sub.textContent = o.sub;
      row.appendChild(sub);
    }
    for (const b of (o.badges || [])) {
      const s = document.createElement("span");
      s.className = "st " + (b.cls || "");
      s.textContent = b.text;
      row.appendChild(s);
    }
    if (o.value === cur) {
      const tick = document.createElement("span");
      tick.className = "pi-sub";
      tick.style.marginLeft = "auto";
      tick.textContent = "✓";
      row.appendChild(tick);
    }
    row.onclick = () => pickPicker(i);
    box.appendChild(row);
  });
}

/** Commit the option at `index` in the filtered list. */
function pickPicker(index) {
  const id = pickerState.openId;
  const opt = pickerState.filtered[index];
  if (!id || !opt) return;
  const hidden = document.getElementById(id);
  hidden.value = opt.value;
  setPickerOptions(id, pickerState.registry[id] || []);
  closePicker();
  // Fire the same event a native select would, so existing handlers keep working.
  hidden.dispatchEvent(new Event("change"));
}

// Wiring: one delegated listener covers every picker button.
document.addEventListener("click", (e) => {
  const btn = e.target.closest && e.target.closest(".picker-btn");
  if (btn) { e.preventDefault(); openPicker(btn.dataset.picker); }
});
document.getElementById("picker-cancel").onclick = closePicker;
document.getElementById("picker-search").addEventListener("input", (e) => {
  pickerState.activeIndex = 0;
  renderPickerList(e.target.value);
});
document.getElementById("picker-search").onkeydown = (e) => {
  if (e.key === "ArrowDown") { e.preventDefault(); pickerState.activeIndex = Math.min(pickerState.activeIndex + 1, pickerState.filtered.length - 1); renderPickerList(e.target.value); }
  else if (e.key === "ArrowUp") { e.preventDefault(); pickerState.activeIndex = Math.max(pickerState.activeIndex - 1, 0); renderPickerList(e.target.value); }
  else if (e.key === "Enter") { e.preventDefault(); pickPicker(pickerState.activeIndex); }
  else if (e.key === "Escape") { e.preventDefault(); closePicker(); }
};

/**
 * Populate the playground's account and model selectors.
 *
 * The model list comes from the PROBE results when a probe has run, not from
 * `config.models`. `config.models` is the operator's manual allowlist and is
 * usually a single entry, so the dropdown showed one option even when the
 * accounts could serve a dozen — which is what made it feel broken. Probing is
 * the honest source: it reflects what the pool can actually call.
 */
async function loadPlaygroundAccounts() {
  try {
    const [accR, cfgR, probeR] = await Promise.all([
      api("/accounts"),
      api("/config"),
      api("/models/probe"),
    ]);
    const acc = (await accR.json()).accounts || [];
    const cfg = await cfgR.json();
    const probe = await probeR.json().catch(() => ({}));

    // Accounts: "auto round-robin" first, then one row per account with its
    // provider, plan and live status so the choice is informed.
    setPickerOptions("pg-account", [
      { value: "", label: T("auto_rr"), sub: "" },
      ...acc.map(a => ({
        value: a.id,
        label: a.name,
        sub: a.provider + " · " + (a.plan === "start-plan" ? T("plan_free") : T("plan_paid")),
        badges: [{ text: T("st_" + a.status), cls: a.status }],
      })),
    ]);

    // Models: probe results when available, else the configured allowlist.
    let models = Array.isArray(probe.union) && probe.union.length ? probe.union : null;
    if (!models) models = (cfg.models && cfg.models.length ? cfg.models : ["glm-5.3-flash"]);
    setPickerOptions("pg-model", models.map(m => ({ value: m, label: m, sub: "" })));
    // Restore the last choice BEFORE falling back to the first entry, and only
    // accept it if it is still offered: the model list comes from probe results,
    // so a stored model can disappear when an account is removed or upstream
    // stops serving it. A stale value would leave the picker captioned with a
    // model the pool cannot call.
    const sel = pgLoadSelection();
    if (!document.getElementById("pg-model").value && sel.model && models.includes(sel.model)) {
      setPickerValue("pg-model", sel.model);
    }
    if (!document.getElementById("pg-model").value && models.length) {
      setPickerValue("pg-model", models[0]);
    }
    // Same for the account, matched against the ids actually in the pool.
    if (!document.getElementById("pg-account").value && sel.account
        && acc.some((a) => a.id === sel.account)) {
      setPickerValue("pg-account", sel.account);
    }
    // Remember what the picker settled on WITHOUT treating it as a switch: this
    // is the baseline the first real change is compared against, and a fresh
    // conversation has nothing to switch away from. Skipped when the operator
    // already has a selection, so re-entering the page does not reset it.
    if (!pgLastModel) pgLastModel = document.getElementById("pg-model").value;
    const hint = document.getElementById("pg-model-src");
    if (hint) hint.textContent = (Array.isArray(probe.union) && probe.union.length)
      ? T("pg_model_probed").replace("{n}", String(models.length))
      : T("pg_model_configured");
  } catch {}
}
function chatBubble(cls, text) {
  const sc = document.getElementById("pg-chat");
  const hint = sc.querySelector(".chat-empty");
  if (hint) hint.remove();
  const d = document.createElement("div");
  d.className = "msg " + cls; d.textContent = text;
  sc.appendChild(d); sc.scrollTop = sc.scrollHeight;
  return d;
}
async function pgSend() {
  const input = document.getElementById("pg-input");
  const text = input.value.trim();
  if (!text) return;
  input.value = "";
  chatBubble("user", text);
  pgHistory.push({ role: "user", content: text });
  pgSave();
  const bot = chatBubble("bot", "…");
  try {
    const resp = await api("/chat", {
      method: "POST",
      body: JSON.stringify({
        model: document.getElementById("pg-model").value,
        accountId: document.getElementById("pg-account").value || undefined,
        // Only real dialogue goes upstream. The saved conversation also holds
        // `error` rows and model-switch `note` dividers (both are what the
        // operator wants to still read after a refresh), and the model API
        // accepts only user/assistant.
        messages: pgHistory.filter((m) => m.role === "user" || m.role === "assistant"),
        stream: true,
      }),
    });
    if (!resp.ok) {
      const t = await resp.text();
      bot.remove();
      const errText = "HTTP " + resp.status + " · " + t.slice(0, 220);
      chatBubble("err", errText);
      // Recorded, not just displayed: the failure text is the evidence of what
      // happened, and it is what the operator looks for after a refresh.
      //
      // The user's message is NOT popped here. It used to be, on the reasoning
      // that a message the engine never answered should not be replayed — but it
      // stays on screen (the bubble is drawn before the request is sent), so
      // dropping it made the saved conversation disagree with the visible one.
      // Worse, while upstream was blocking every send failed, so each attempt
      // saved the message and then immediately saved an empty list over it:
      // nothing ever survived a refresh, reported as "chat history does not
      // work". Re-sending it next time is what a person re-asking would do.
      pgHistory.push({ role: "error", content: errText });
      pgSave();
      return;
    }
    const reader = resp.body.getReader(), dec = new TextDecoder();
    let buf = "", think = "", out = "";
    bot.textContent = "";
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buf += dec.decode(value, { stream: true });
      let i;
      while ((i = buf.indexOf("\n")) >= 0) {
        const line = buf.slice(0, i).trim(); buf = buf.slice(i + 1);
        if (!line.startsWith("data:")) continue;
        const p = line.slice(5).trim();
        if (p === "[DONE]") continue;
        try {
          const d = (JSON.parse(p).choices || [{}])[0].delta || {};
          if (d.reasoning_content) { think += d.reasoning_content; }
          if (d.content) { out += d.content; bot.textContent = out; }
        } catch {}
      }
    }
    if (think) {
      const t = document.createElement("div"); t.className = "think"; t.textContent = think.slice(0, 800);
      bot.prepend(t);
    }
    if (!out && !think) bot.textContent = "(empty)";
    pgHistory.push({ role: "assistant", content: out || "(empty)" });
    pgSave();
  } catch (e) {
    bot.remove(); chatBubble("err", String(e));
    // Same as the non-OK branch: the failure is recorded, and the message that
    // is still on screen is kept.
    pgHistory.push({ role: "error", content: String(e) });
    pgSave();
  }
}

// ── Logs: one merged stream ────────────────────────────────────────────────
// Requests and engine messages used to be two panes, which meant the answer to
// "what happened at 14:03?" required reading both and interleaving them by eye.
// They are now ONE list sorted by time: a request row carries its caller, path,
// model, account, status, TTFB and tokens, and an engine line carries its
// severity. The filters below apply to the merged list.
//
// The list is rebuilt (not appended) on every poll so the two sources stay in
// time order — appending would interleave them by arrival, not by when they
// happened.
let logTimer = null;

/**
 * `YYYY-MM-DD HH:MM:SS` — the full local timestamp for a log line.
 *
 * The year is included because a log line outlives the session that produced
 * it: an exported CSV is read weeks later, and a bare `MM-DD` is ambiguous
 * across a year boundary. Seconds matter for the same reason — the merged
 * timeline interleaves requests with engine messages, and minute precision
 * cannot order events within a burst.
 */
function fmtLogTime(ms) {
  const d = new Date(ms);
  const p = (n) => String(n).padStart(2, "0");
  return d.getFullYear() + "-" + p(d.getMonth() + 1) + "-" + p(d.getDate()) + " " +
         p(d.getHours()) + ":" + p(d.getMinutes()) + ":" + p(d.getSeconds());
}

function renderMergedLog(rows, entries, filter) {
  const box = document.getElementById("log-box");
  const atBottom = box.scrollHeight - box.scrollTop - box.clientHeight < 40;

  // One array of {at, html} then a single sort: merging two already-sorted
  // sources is easy to get subtly wrong, and the cost here is negligible.
  const items = [];
  for (const x of rows) {
    const ok = x.status >= 200 && x.status < 400;
    items.push({
      at: x.started,
      kind: "req",
      search: [x.ip, x.path, x.model, x.accountName, x.userAgent, String(x.status)]
        .filter(Boolean).join(" ").toLowerCase(),
      html:
        '<span class="lg-t">' + fmtLogTime(x.started) + "</span>" +
        '<span class="lg-k lg-k-req">REQ</span>' +
        '<span class="lg-s ' + (ok ? "ok" : "error") + '">' + x.status + "</span>" +
        '<span class="lg-m">' + esc(x.model) + "</span>" +
        '<span class="lg-d">' + esc(shortUa(x.userAgent)) + " · " + esc(x.ip || "—") +
        " · " + esc(x.path || "—") + "</span>" +
        '<span class="lg-a">' + esc(x.accountName || "—") + "</span>" +
        '<span class="lg-x">' + fmtMs(x.ttfbMs) + " · " + (x.tokens || 0) + " tok</span>",
    });
  }
  for (const e of entries) {
    items.push({
      at: e.at || Date.now(),
      kind: "log",
      search: String(e.text || "").toLowerCase(),
      html:
        '<span class="lg-t">' + fmtLogTime(e.at || Date.now()) + "</span>" +
        '<span class="lg-k lg-k-' + esc(e.level || "info") + '">' +
        esc((e.level || "info").toUpperCase()) + "</span>" +
        '<span class="lg-d lg-d-full">' + esc(e.text) + "</span>",
    });
  }
  items.sort((a, b) => a.at - b.at);

  const want = filter.kind || "all";
  const q = (filter.q || "").trim().toLowerCase();
  const shown = items.filter((it) => {
    if (want !== "all" && it.kind !== want) return false;
    if (q && !it.search.includes(q)) return false;
    return true;
  });

  document.getElementById("rq-count").textContent =
    T("rq_count").replace("{n}", String(shown.length)).replace("{m}", String(items.length));

  if (!shown.length) {
    box.innerHTML = '<div class="bal-empty">' + T("rq_none") + "</div>";
    return;
  }
  // innerHTML in one assignment: a node per line plus a node per poll leaks
  // DOM, and the engine's ring buffer already bounds how much history exists.
  box.innerHTML = shown.map((it) => '<div class="logline">' + it.html + "</div>").join("");
  if (atBottom) box.scrollTop = box.scrollHeight;
}

/**
 * Fetch both sources and merge.
 *
 * The engine log is fetched INCREMENTALLY (cursor) while the request list is a
 * full snapshot; the accumulated engine lines are kept in `logEntries` so the
 * merged view can re-sort them against the newest request rows.
 */
let logSince = 0;
let logEntries = [];
async function pollLogs() {
  try {
    const [rq, lg] = await Promise.all([
      api("/requests?limit=200"),
      api("/logs?since=" + logSince),
    ]);
    const rd = await rq.json();
    const ld = await lg.json();
    logSince = ld.nextSince ?? logSince;
    const fresh = ld.entries && ld.entries.length
      ? ld.entries
      : (ld.lines || []).map((t) => ({ text: t, level: "info", at: Date.now() }));
    if (fresh.length) {
      logEntries = logEntries.concat(fresh);
      // Bound the merged history. Without this the array grows for as long as
      // the panel stays open, and every poll re-sorts all of it.
      if (logEntries.length > 2000) logEntries = logEntries.slice(-2000);
    }
    renderMergedLog(rd.rows || [], logEntries, {
      kind: document.getElementById("rq-kind").value || "all",
      q: document.getElementById("rq-search").value,
    });
  } catch (e) {
    // Report rather than swallow. A bare `catch {}` here hid a ReferenceError
    // for a whole debugging round: the page simply showed an empty log, with
    // nothing to distinguish "no data" from "the renderer is broken".
    const box = document.getElementById("log-box");
    if (box) box.innerHTML = '<div class="bal-empty">' + esc(T("log_err") + " " + e) + "</div>";
  }
}

document.getElementById("rq-kind").addEventListener("change", () => pollLogs());
document.getElementById("rq-status").addEventListener("change", () => pollLogs());
document.getElementById("rq-account").addEventListener("change", () => pollLogs());
document.getElementById("rq-refresh").onclick = () => pollLogs();

logTimer = setInterval(() => {

/**
 * Countdown ticker for every `[data-until]` element.
 *
 * One interval for the whole page rather than a timer per element: the balance
 * list and the account cards both render countdowns, and both re-render on every
 * poll, so per-element timers would leak one per repaint. Formatting is
 * `Dd HH:MM:SS` / `HH:MM:SS`, dropping the leading zero units.
 */
function fmtCountdown(ms) {
  const total = Math.max(0, Math.floor(ms / 1000));
  const d = Math.floor(total / 86400);
  const h = Math.floor((total % 86400) / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  const p = (n) => String(n).padStart(2, "0");
  if (d > 0) return d + "d " + p(h) + ":" + p(m) + ":" + p(s);
  if (h > 0) return p(h) + ":" + p(m) + ":" + p(s);
  return p(m) + ":" + p(s);
}
function tickCountdowns() {
  const now = Date.now();
  for (const el of document.querySelectorAll("[data-until]")) {
    const until = Number(el.dataset.until);
    if (!Number.isFinite(until)) continue;
    const left = until - now;
    // `data-fmt="days"` is for plan windows measured in days: the ticker is still
    // second-accurate, but "3d 23:59:59" is noise for a deadline that far out, so
    // those show "3天 23小时". Anything shorter than a day falls back to the clock
    // reading, which is where the seconds start to matter.
    const long = el.dataset.fmt === "days" && left > 86_400_000;
    if (long) {
      const d = Math.floor(left / 86_400_000);
      const h = Math.floor((left % 86_400_000) / 3_600_000);
      el.textContent = T("cd_days").replace("{d}", String(d)).replace("{h}", String(h));
    } else {
      el.textContent = left > 0 ? fmtCountdown(left) : T("cd_expired");
    }
    // Reuse the login dialog's thresholds so "running low" reads the same
    // everywhere: amber inside a minute, struck through once past. A plan window
    // gets a much longer lead time — a day's notice, not a minute's, because
    // losing the daily allowance is not something to discover after the fact.
    const warnAt = el.dataset.fmt === "days" ? 86_400_000 : 60_000;
    el.classList.toggle("cd-warn", left <= warnAt && left > 0);
    el.classList.toggle("cd-dead", left <= 0);
    // Colour the whole row, so an imminently-expiring plan is visible without
    // reading the clock.
    const row = el.closest(".plan-exp");
    if (row) row.classList.toggle("soon", left <= warnAt && left > 0);
  }
}
setInterval(tickCountdowns, 1000);

/** Render the stored probe detail for one account, if a probe has run. */
async function renderStoredProbe(accountId) {
  if (!accountId) return;
  try {
    const d = await (await api("/models/probe")).json();
    const rec = (d.records || []).find(r => r.accountId === accountId);
    const stat = document.getElementById("md-stat");
    if (!rec) {
      // No stored result for this account: probe it now rather than leaving the
      // page blank.
      //
      // Waiting for the scheduler is not good enough — it runs every 6 hours,
      // and an account added since the last sweep has no record at all, so the
      // page looked empty with no indication that anything was pending. Guarded
      // so re-selecting the same account does not fire a second sweep, and the
      // server refuses a concurrent one anyway.
      document.getElementById("md-result").innerHTML = "";
      stat.textContent = T("md_probing");
      if (!probeAutoStarted.has(accountId)) {
        probeAutoStarted.add(accountId);
        runProbe(accountId).catch(() => { probeAutoStarted.delete(accountId); });
      }
      return;
    }
    stat.textContent = T("md_last").replace("{t}", new Date(rec.probedAt).toLocaleString()) +
      " · " + rec.okModels.length + " / " + rec.detail.length;
    renderProbeRows(rec.detail);
  } catch {}
}
/** Accounts already auto-probed this session, so a re-select does not re-fire. */
const probeAutoStarted = new Set();

/** Render the per-model result rows. */
function renderProbeRows(detail) {
  const el = document.getElementById("md-result");
  el.innerHTML = "";
  for (const r of (detail || [])) {
    const row = document.createElement("div");
    row.className = "set-row";
    // The note carries the distinction the badge cannot: a verdict measured in
    // this run versus one carried forward from an earlier one because this run
    // was blocked, and how many times. Without it "ok" covered both a model that
    // had just answered and one the upstream had been refusing for hours.
    const stale = r.carriedRuns ? ' <span style="color:var(--gold)">(' + r.carriedRuns + T("md_carried") + ')</span>' : "";
    row.innerHTML = '<div class="lbl" style="font-family:var(--mono)">' + esc(r.id) + "</div>" +
      '<span class="st ' + (r.ok ? "ok" : "paused") + '">' + (r.ok ? T("md_ok") : T("md_no")) + "</span>" +
      (r.note ? ' <span class="sub" style="margin:0">' + esc(r.note) + "</span>" : "") + stale;
    el.appendChild(row);
  }
}

/**
 * Poll the probe run while it is in flight and drive the progress bar.
 *
 * A full sweep is one request per model per account, so it takes tens of
 * seconds. Without this the button looked dead for the whole run.
 */
let probePoll = null;
function startProbeProgress() {
  const wrap = document.getElementById("md-progress");
  const bar = document.getElementById("md-progress-bar");
  const stat = document.getElementById("md-stat");
  wrap.style.display = "block";
  bar.classList.add("indet");
  if (probePoll) clearInterval(probePoll);
  probePoll = setInterval(async () => {
    try {
      const d = await (await api("/models/probe")).json();
      const s = d.state || {};
      if (s.running) {
        const pct = s.total ? Math.round((s.done / s.total) * 100) : 0;
        bar.classList.remove("indet");
        bar.style.width = pct + "%";
        stat.textContent = T("md_progress")
          .replace("{done}", String(s.done)).replace("{total}", String(s.total))
          + (s.current ? " · " + s.current : "");
      } else {
        stopProbeProgress();
        await refreshProbeView();
      }
    } catch { /* transient — the next tick retries */ }
  }, 1200);
}
function stopProbeProgress() {
  if (probePoll) { clearInterval(probePoll); probePoll = null; }
  const bar = document.getElementById("md-progress-bar");
  bar.classList.remove("indet");
  bar.style.width = "0%";
  document.getElementById("md-progress").style.display = "none";
}

/** Re-read stored results and repaint the page. */
async function refreshProbeView() {
  const d = await (await api("/models/probe")).json();
  renderProbeSummary(d);
  await renderStoredProbe(document.getElementById("md-account").value);
  loadPlaygroundAccounts();
}

/** Header line: whether auto is armed, and the served model count. */
function renderProbeSummary(d) {
  const note = document.getElementById("md-auto-note");
  const n = (d.records || []).length;
  const union = d.union || [];
  const hours = d.intervalMs ? Math.round(d.intervalMs / 3600000) : 0;
  const s = d.state || {};
  // A sweep where every account was blocked (captcha / risk control) measured
  // nothing — say so, rather than implying the model list is up to date.
  const blocked = !s.running && s.skipped > 0
    ? " · " + T("md_blocked").replace("{n}", String(s.skipped))
    : "";
  note.textContent = (d.auto
    ? T("md_auto_on").replace("{h}", String(hours))
    : T("md_auto_off"))
    + (n ? " · " + T("md_union").replace("{n}", String(union.length)) : "")
    + blocked;
}

/**
 * Probe one account and render the result.
 *
 * Extracted from the button handler so the page can also fire it automatically
 * (see renderStoredProbe): an account with no stored record otherwise showed a
 * blank page until the operator found the button or the 6-hour sweep ran.
 */
async function runProbe(accountId) {
  const stat = document.getElementById("md-stat");
  const d = await (await api("/models/probe", { method: "POST", body: JSON.stringify({ accountId }) })).json();

  renderProbeRows(d.detail);
  // The probe writes back to /v1/models, so refresh anything that lists models
  // from it — otherwise the change is invisible until a page reload.
  loadPlaygroundAccounts();
  renderProbeSummary(await (await api("/models/probe")).json());
}

async function loadSettings() {
  try {
    const cfg = await (await api("/config")).json();
    document.getElementById("set-apiurl").value = location.origin + "/v1";
    // Full key from the server (the sessionStorage copy can be a partial paste).
    document.getElementById("set-key").value = cfg.proxyApiKey || "";
    // The baseline the dirty check compares against. Set from the server's
    // answer, not from the input, so the two cannot drift.
    storedKey = (cfg.proxyApiKey || "").trim();
    // The save button starts inert: the field already holds the stored key, so
    // there is nothing to save until the operator changes it. Without this the
    // button sat there looking actionable, and pressing it asked to confirm a
    // write of the value that was already stored.
    markKeyDirty(false);
    if (cfg.pool) {
      document.getElementById("set-gate").value = cfg.pool.maxConcurrentPerAccount;
      document.getElementById("set-cooldown").value = Math.round(cfg.pool.cooldownMs / 1000);
      if (typeof cfg.pool.modelCooldownMs === "number") {
        document.getElementById("set-model-cd").value = Math.round(cfg.pool.modelCooldownMs / 1000);
      }
      document.getElementById("set-spacing").value = cfg.pool.minSpacingMs;
      // Model-level ceilings. Rendered as `model=limit` lines so the field shows
      // exactly what the pool is applying, including the glm-5.3 override that
      // exists by default — otherwise an operator who lowers it by hand has no
      // way to see (or undo) the built-in value.
      const pm = cfg.pool.maxConcurrentPerModel || {};
      if (typeof pm.default === "number") document.getElementById("set-gate-model").value = pm.default;
      const over = Object.entries(pm.byModel || {}).map(([m, n]) => m + "=" + n);
      document.getElementById("set-gate-over").value = over.join("\n");
      if (typeof cfg.pool.overflowFactor === "number") {
        document.getElementById("set-overflow").value = cfg.pool.overflowFactor;
      }
    }
    if (cfg.claim) {
      claimAuto = !!cfg.claim.auto;
      const b = document.getElementById("set-claim-auto");

      document.getElementById("set-proxy-noproxy").value = cfg.proxy.noProxy || "";

    }
    if (cfg.panel) {
      document.getElementById("set-idle").value = cfg.panel.idleTimeoutMinutes;
      renderIdleStat(cfg.panel.idleTimeoutMinutes);
    }
  } catch {}
  // Current build version for the update row (the overview endpoint carries it).
  try {
    const ov = await (await api("/overview")).json();
    if (ov.version) document.getElementById("upd-cur").textContent = ov.version;
  } catch {}

  // perform a check, so opening Settings repeatedly costs no GitHub quota.
  try {
    const d = await (await api("/update/last")).json();
    if (d.checkedAt && d.result) renderUpdateCheck(d.result, d.checkedAt);
  } catch {}
}

/**
 * Render an update-check result into the Settings row.
 *
 * Shared by the manual button and the on-load read of the background result, so
 * the two cannot drift into showing different things for the same answer.
 */
function renderUpdateCheck(d, checkedAt) {
  const latest = document.getElementById("upd-latest");
  const actions = document.getElementById("upd-action-row");
  if (d.currentVersion) document.getElementById("upd-cur").textContent = d.currentVersion;
  if (d.error === "no_release") {
    latest.textContent = T("upd_no_release");
    actions.style.display = "none";
    return;
  }
  if (d.error) {
    latest.textContent = T("upd_err") + " " + d.error;
    actions.style.display = "none";
    return;
  }
  if (d.updateAvailable) {
    // The age matters: a result from the 5-day timer is not "checked just now",
    // and saying so would overstate how fresh the news is.
    const when = checkedAt ? " · " + T("upd_checked") + " " + fmtLocalTime(Math.floor(checkedAt / 1000)) : "";
    latest.textContent = T("upd_avail") + " " + d.latestVersion + when;
    actions.style.display = "";
    const notes = document.getElementById("upd-notes");
    if (d.releaseUrl) { notes.href = d.releaseUrl; notes.style.display = ""; }
    // The engine only CHECKS for updates now. Downloading and swapping the
    // binary in place was removed: it needed a restart script, a staged file
    // and a running-process swap to work, and every failure mode in that chain
    // (a blocked download, a 403 from the API, an unsigned archive) left the
    // operator with a half-applied update and no clear recovery. Pointing at
    // the release page instead is one click for the operator and cannot leave
    // the install in a state the engine has to explain.
    const goto = document.getElementById("upd-goto");
    if (d.releaseUrl) { goto.href = d.releaseUrl; goto.style.display = ""; }
  } else {
    latest.textContent = T("upd_current") + (d.latestVersion ? " (" + d.latestVersion + ")" : "");
    actions.style.display = "none";
  }
}

/**
 * Download a file and VERIFY it arrived whole.
 *
 * These were plain `<a href>` links, which is the simplest thing that works and
 * is exactly why a truncated download went unnoticed: the browser saves
 * whatever it received and nothing compares it to what the server sent. The
 * download is now fetched, checked against `Content-Length` and the declared
 * byte counts, and only then handed to the browser as a blob.
 *
 * The filename still comes from Content-Disposition, so the saved name is the
 * same as before.
 */
async function downloadVerified(url, label) {
  let resp;
  try {
    resp = await fetch(url, { credentials: "same-origin" });
  } catch (e) {
    toast(T("dl_net") + " " + String(e));
    return;
  }
  if (!resp.ok) {
    toast(T("dl_fail").replace("{s}", String(resp.status)));
    return;
  }
  const declared = Number(resp.headers.get("content-length") || "0");
  let text;
  try {
    text = await resp.text();
  } catch (e) {
    // A body that dies mid-stream lands here — the most common form of a
    // truncated download, and the one a plain link hides completely.
    toast(T("dl_truncated") + " " + String(e));
    return;
  }
  const got = new Blob([text]).size;
  if (declared > 0 && got !== declared) {
    toast(T("dl_short").replace("{g}", String(got)).replace("{d}", String(declared)));
    return;
  }
  // For JSON bundles, also check the per-file byte counts the server recorded.
  if (url.endsWith(".json")) {
    try {
      const parsed = JSON.parse(text);
      const bad = (parsed.files || []).filter(
        (f) => typeof f.bytes === "number" && new Blob([f.content]).size !== f.bytes,
      );
      if (bad.length) {
        toast(T("dl_badfiles").replace("{n}", bad.map((f) => f.name).join(", ")));
        return;
      }
    } catch (e) {
      toast(T("dl_badjson") + " " + String(e));
      return;
    }
  }
  const name = filenameFrom(resp.headers.get("content-disposition")) || label;
  const a = document.createElement("a");
  a.href = URL.createObjectURL(new Blob([text], { type: resp.headers.get("content-type") || "application/octet-stream" }));
  a.download = name;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(a.href), 10000);
  toast(T("dl_ok").replace("{n}", name).replace("{b}", fmtBytes(got)));
}

/** Pull `filename="..."` out of a Content-Disposition header. */
function filenameFrom(cd) {
  if (!cd) return "";
  const m = /filename\*?=(?:UTF-8'')?"?([^";]+)"?/i.exec(cd);
  return m ? decodeURIComponent(m[1]) : "";
}

function fmtBytes(n) {
  if (n < 1024) return n + " B";
  if (n < 1024 * 1024) return (n / 1024).toFixed(1) + " KB";
  return (n / 1024 / 1024).toFixed(2) + " MB";
}

  //
  // The bundle carries the accounts file, but that file is encrypted with a key

// ── Login gate ────────────────────────────────────────────────────────────────
// The panel's markup ships in the page, so an unauthenticated viewer could
// otherwise read the layout, section names and sidebar links straight from the
// DOM. The shell therefore starts hidden and only the login dialog is shown;
// nothing else is fetched or polled until a session exists.
//
// On load the session cookie is probed FIRST. The cookie is HttpOnly, so script
// cannot see it — the only way to know whether a session survives is to ask the
// server. Skipping that probe is what made every refresh show the login dialog
// even though a valid session was still held: the panel simply never checked.
async function resumeSession() {
  try {
    const r = await fetch("/admin/api/session", { credentials: "same-origin" });
    if (!r.ok) return false;
    const d = await r.json();
    if (!d.ok) return false;
    CSRF = d.csrf || "";
    sessionStorage.setItem("zk-csrf", CSRF);
    return true;
  } catch (e) {
    return false;
  }
}

async function boot() {
  // The markup already carries `gated`, so the panel is hidden before this runs.
  // Kept here as well because boot() re-enters after a failed login attempt, and
  // the gate must still be on at that point.
  document.body.classList.add("gated");
  if (!(await resumeSession())) {
    const key = await askAdminKey();
    if (!key) return;                       // left the gate: stay blank
    if (!(await doLogin(key))) {
      // The dialog has already closed; reopen it so a failed attempt can be
      // retried rather than leaving a blank page with no way forward.
      return boot();
    }
  }
  // Only now is the panel revealed. Nothing above this line may un-hide it.
  document.body.classList.remove("gated");
  // Restore the page the user was on. Done before the data loads so the right
  // page is visible immediately rather than after Overview flashes.
  showPage(lastPage());
  refreshOverview();
  loadBalances();
  refreshAccounts();
  loadSettings();
  pollLogs();
  setInterval(refreshOverview, 5000);
}
boot();
